# 1. Модули (C:\xampp\apache\conf\httpd.conf, раскомментировать)
LoadModule proxy_module      modules/mod_proxy.so
LoadModule proxy_http_module modules/mod_proxy_http.so

# 2. Статика фронта
Alias /gr "C:/xampp/htdocs/gr"
<Directory "C:/xampp/htdocs/gr">
    Options -Indexes
    AllowOverride None
    Require all granted
    DirectoryIndex index.html
    # Чтобы после деплоя не отдавался старый бандл
    <FilesMatch "index\.html$">
        Header set Cache-Control "no-cache, must-revalidate"
    </FilesMatch>
</Directory>

# 3. API
